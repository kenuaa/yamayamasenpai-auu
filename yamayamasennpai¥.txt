    <!-- Start menu -->
    <div id="menu">
        <h2>A new kind of gay game</h2>
        <p>Click the targets falling from above, starting with the one at the very bottom!</p>
        <button onclick="startGame()">ゲーム開始</button>
    </div>

    <!-- Result screen -->
    <div id="result" style="display: none;">
        <h2>Game Over！</h2>
        <p>score: <span id="final-score">0</span></p>
        <p>CPS: <span id="cps">0.00</span></p>
        <button onclick="restartGame()">リトライ</button>
    </div>

    <!-- Game Over -->
    <div id="game-container"></div>

    <script>
        const container = document.getElementById('game-container');
        const timeDisplay = document.getElementById('time');
        const scoreDisplay = document.getElementById('score');
        const menu = document.getElementById('menu');
        const result = document.getElementById('result');
        
        let score = 0;
        let timeLeft = 20;
        let gameInterval;
        let spawnInterval;
        let targets = [];
        let isPlaying = false;

        // Start Game
        function startGame() {
            menu.style.display = 'none';
            result.style.display = 'none';
            score = 0;
            timeLeft = 20;
            targets = [];
            container.innerHTML = '';
            scoreDisplay.textContent = score;
            timeDisplay.textContent = timeLeft;
            isPlaying = true;

            // Start timer (every 1 second)
            gameInterval = setInterval(() => {
                timeLeft--;
                timeDisplay.textContent = timeLeft;
                if (timeLeft <= 0) {
                    endGame();
                }
            }, 1000);

            // Periodically generate targets (initially, generate several at once)
            spawnTarget();
            spawnTarget();
            spawnInterval = setInterval(spawnTarget, 800);
        }

        // Generation of targets
        function spawnTarget() {
            if (!isPlaying) return;

            const target = document.createElement('div');
            target.className = 'target';
            target.textContent = 'Click here';
            
            // Randomly determine the horizontal position
            const maxX = container.clientWidth - 90;
            const randomX = Math.floor(Math.random() * maxX);
            target.style.left = `${randomX}px`;
            
            // Initially placed at the top of the screen
            let posY = 0;
            target.style.top = `${posY}px`;

            container.appendChild(target);
            
            // Managed via an array (stores positional information)
            const targetObj = { element: target, y: posY };
            targets.push(targetObj);

            // Animation loop that moves the target downward
            const moveSpeed = 2 + Math.random() * 2; // Make the falling speed random
            function fall() {
                if (!isPlaying || !targets.includes(targetObj)) return;
                
                targetObj.y += moveSpeed;
                target.style.top = `${targetObj.y}px`;

                // If an item passes the line at the very bottom of the screen (the judgment line), it is removed, and a penalty is applied.
                if (targetObj.y > container.clientHeight - 80) {
                    removeTarget(targetObj);
                    // Perform operations such as reducing the score here, if necessary.
                } else {
                    requestAnimationFrame(fall);
                }
            }
            requestAnimationFrame(fall);

            // Registering a click event
            target.addEventListener('mousedown', () => {
                // A check for the rule where only the lowest target (the one with the maximum Y-coordinate) is considered the correct answer.
                const lowestTarget = targets.reduce((max, t) => t.y > max.y ? t : max, targets[0]);
                
                if (targetObj === lowestTarget) {
                    score++;
                    scoreDisplay.textContent = score;
                    removeTarget(targetObj);
                }
            });
        }

        // Delete target
        function removeTarget(targetObj) {
            if (targetObj.element.parentNode) {
                targetObj.element.parentNode.removeChild(targetObj.element);
            }
            targets = targets.filter(t => t !== targetObj);
        }

        // Game Over
        function endGame() {
            isPlaying = false;
            clearInterval(gameInterval);
            clearInterval(spawnInterval);
            
            document.getElementById('final-score').textContent = score;
            // Calculate CPS over 20 seconds
            document.getElementById('cps').textContent = (score / 20).toFixed(2);
            
            result.style.display = 'block';
        }

        // retry
        function restartGame() {
            startGame();
        }
    </script>
